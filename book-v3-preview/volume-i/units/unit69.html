本单元的"该学的定理"：

  • 取用子：x ∈ ⋃F ↔ ∃A∈F, x ∈ A ✓ · x ∈ ⋂F ↔ ∀A∈F, x ∈ A ✓（两条顶层具名 ✓）；
  • 旗舰一：⋃F ⊆ A ↔ ∀B∈F, B ⊆ A ✓（"并是最小上界" ✓）；
  • 旗舰二：A ⊆ ⋂F ↔ ∀B∈F, A ⊆ B ✓（"交是最大下界" ✓）；
  • A ∈ F ⇒ ⋂F ⊆ A ✓（下界性 ✓）

theorem sUnion_subset_iff (α : Type) (F : Set (Set α)) (A : Set α) :
    (⋃₀ F) ⊆ A ↔ ∀ (B : Set α), B ∈ F → B ⊆ A :=
  @Iff.intro ((⋃₀ F) ⊆ A) (∀ (B : Set α), B ∈ F → B ⊆ A)
    (fun (h : (⋃₀ F) ⊆ A) =>
      fun (B : Set α) (hB : B ∈ F) =>
        fun (b : α) (hb : b ∈ B) =>
          h b ((fun (hu : b ∈ (⋃₀ F) ↔ ∃ (C : Set α), C ∈ F ∧ b ∈ C) =>
            Iff.mpr hu (Exists.intro B
              (And.intro hB hb))) (Set.mem_sUnion_iff α F b)))
    (fun (h : ∀ (B : Set α), B ∈ F → B ⊆ A) =>
      fun (x : α) (hx : x ∈ (⋃₀ F)) =>
        Set.mem_sUnion_elim α F x (x ∈ A) hx
          (fun (B : Set α) (hB : B ∈ F) (hxB : x ∈ B) => h B hB x hxB))

    A ⊆ (⋂₀ F) ↔ ∀ (B : Set α), B ∈ F → A ⊆ B :=
