theorem mem_sdiff_of_mem_not_mem (α : Type) (x : α) (A : Set α) (B : Set α)
    (hx : x ∈ A) (hnb : ¬ (x ∈ B)) : x ∈ (A \ B) :=
  (fun (hm : x ∈ (A \ B) ↔ (x ∈ A ∧ ¬ (x ∈ B))) =>
    Iff.mpr hm (And.intro hx hnb)) (Set.mem_sdiff x A B)

theorem not_mem_of_eq_empty (α : Type) (x : α) (C : Set α) (h : C = ∅)
    (hx : x ∈ C) : False :=
  (fun (hu : x ∈ ∅ ↔ False) => Iff.mp hu (@Eq.subst.{1} (Set α) (fun (D : Set α) => x ∈ D)

theorem subset_of_sdiff_empty (α : Type) (A : Set α) (B : Set α) (h : A \ B = ∅) :
    A ⊆ B :=
  fun (x : α) (hx : x ∈ A) =>
    em_cases (x ∈ B) (x ∈ B) (fun (hb : x ∈ B) => hb)
      (fun (hnb : ¬ (x ∈ B)) =>
        @False.elim (x ∈ B)
          (not_mem_of_eq_empty α x (A \ B) h (mem_sdiff_of_mem_not_mem α x A B hx hnb)))

theorem not_forall_imp_exists (α : Type) (P : α → Prop) (h : ¬ (∀ (x : α), ¬ (P x))) :
    ∃ (x : α), P x :=
  Classical.byContradiction (∃ (x : α), P x)
    (fun (hne : ¬ (∃ (x : α), P x)) =>
      h (fun (x : α) (hx : P x) => hne (Exists.intro x hx)))
编写注记 · 教材对照与课程边界（5 条）

靶子（教材）：Enderton《Elements of Set Theory》§1.2 / §1.3（等价的证明方式）·Halmos《Naive Set Theory》§1。大纲 §4.1（经典逻辑线开启 ✓）。

⚠⚠ 本单元的背景（课程史上的一个转折 ✓）：本课此前是构造性的 ✓ ⇒ 手册里明令不得写 ¬¬P ⇒ P、A \ B = ∅ ⇒ A ⊆ B、¬∀ ⇒ ∃ 这类经典等价 ✗（单元57 就因此删掉了 A \ B = ∅ ⇒ A ⊆ B ✓，我当时还把它记成"本课是构造性的"✓）。现在内核线落地了 Classical.em（G-74 排中律） ✓ ⇒ 这些定理可以证了 ✓ ⇒ 本单元把当年删掉的那条请回来 ✓，并系统给出经典推论的三件套 ✓。

⚠ 写法纪律（本轮新增 ✓）：False.elim 要把目标写全 ✓（False.elim (x ∈ B) … ✓）；含于空集的成员是矛盾要经 Eq.subst（集合层替换 ✓）+ Set.mem_empty_iff_false ✓ ⇒ 抽成小引理 not_mem_of_eq_empty ✓（先拆链 ✓）。

本单元只允许用：intro / exact / apply / constructor / cases / have，加 import 进来的库引理与 prelude 里的 Classical.* ✓。本单元 import 三样：lib.Set · lib.Rel · lib.Order。
