本单元的"该学的定理"：

  • 取用子 与 打包子（LeOn ✓）· 左逆 ⇒ 单射（限制在 A 上） ✓ ·右逆 ⇒ 单射（对偶 ✓）—— 两条都要一次 Eq.subst 把 g (f x) 沿 f x = f y 搬成 g (f y) ✓（格位表第二行 ✓）；

theorem injOn_of_leftInv (α : Type) (β : Type) (f : α → β) (g : β → α) (A : Set α)
    (hgf : Set.LeftInvOn α β g f A) : Set.InjOn α β f A :=
  fun (x : α) (hx : x ∈ A) (y : α) (hy : y ∈ A) (h : f x = f y) =>
    @Eq.trans.{1} α x (g (f x)) y (@Eq.symm.{1} α (g (f x)) x (hgf x hx))
      (@Eq.trans.{1} α (g (f x)) (g (f y)) y
        (@Eq.subst.{1} β (fun (v : β) => g (f x) = g v) (f x) (f y) h
          (@Eq.refl.{1} α (g (f x))))
        (hgf y hy))

theorem leOn_of_equiv (α : Type) (β : Type) (A : Set α) (B : Set β) (h : A ≈ B) :
    LeOn α β A B ∧ LeOn β α B A :=
  Set.Equiv.elim α β A B (LeOn α β A B ∧ LeOn β α B A) h
    (fun (f : α → β) (g : β → α) (hf : Set.MapsTo α β f A B)
      (hg : Set.MapsTo β α g B A) (hgf : Set.LeftInvOn α β g f A)
      (hfg : Set.RightInvOn α β f g B) =>
      And.intro
        (leOn_intro α β A B f (injOn_of_leftInv α β f g A hgf) hf)
        (leOn_intro β α B A g (injOn_of_rightInv α β g f B hfg) hg))
